cmake_minimum_required(VERSION 3.22)

# Setup compiler settings
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Core project settings
project(juicy_hello_world_cmake
    LANGUAGES C ASM
)
# Create the executable to the project
add_executable(${CMAKE_PROJECT_NAME})

##############################################################################################
#  Compiler additional options (compiler is the interface target of the toolchain script)
##############################################################################################
if(CMAKE_BUILD_TYPE STREQUAL "Release")
    target_compile_options(compiler INTERFACE -flto -O2 -g3)
    target_link_options(compiler INTERFACE -flto)
elseif(CMAKE_BUILD_TYPE STREQUAL "Debug")
    target_compile_options(compiler INTERFACE -flto -Og -g3)
    target_link_options(compiler INTERFACE -flto)
else()
    message(FATAL_ERROR "${CMAKE_BUILD_TYPE} is not a valid build type")
endif()


##############################################################################################
#  STM32 Generated directory related
##############################################################################################
add_subdirectory(cmake/stm32cubemx)
# Add compiler interface to STMCUBEMX generated file
target_link_libraries(STM32_Drivers PRIVATE compiler)
# Change include properties from STM32CubeMX to avoid warnings from these headers
get_target_property(STM32CUBEMX_INCLUDES stm32cubemx INTERFACE_INCLUDE_DIRECTORIES)
target_include_directories(stm32cubemx SYSTEM INTERFACE ${STM32CUBEMX_INCLUDES})
# Remove warnings from files generated by STM32CubeMX
target_compile_options(STM32_Drivers PUBLIC
    # syscalls.c and sysmem.c was generated and got some warnings... Silent them.
    -Wno-missing-prototypes
    -Wno-conversion
    -Wno-sign-conversion
)
# Link to the main executable
target_link_libraries(${CMAKE_PROJECT_NAME} STM32_Drivers)

##############################################################################################
# CMake Custom headers generation (version and configurations for the user application)
##############################################################################################
function(set_target_git_version_generation)
    find_package(Git REQUIRED)
    set(VERSION_HEADER ${CMAKE_BINARY_DIR}/cmake_version.h)
    add_custom_command(
        OUTPUT ${VERSION_HEADER}
        COMMAND ${CMAKE_COMMAND}
            -DOUTPUT_FILE=${VERSION_HEADER}
            -DVERSION_TEMPLATE=${CMAKE_SOURCE_DIR}/cmake/cmake_version.h.in
            -P ${CMAKE_CURRENT_LIST_DIR}/cmake/generate-version.cmake
        COMMENT "Generating version header from Git"
    )
    add_custom_target(git_version_generation ALL DEPENDS ${VERSION_HEADER})
endfunction()

# Add the git target above that will be call at each build
set_target_git_version_generation() 

# Project informations (generated once)
configure_file(
    "${PROJECT_SOURCE_DIR}/cmake/cmake_config.h.in"
    "${CMAKE_BINARY_DIR}/cmake_config.h"
    @ONLY
)
message(NOTICE "Build type: ${CMAKE_BUILD_TYPE}")

##############################################################################################
# Application
##############################################################################################
# Libraries and interfaces
target_link_libraries(${CMAKE_PROJECT_NAME} compiler)
add_dependencies(${CMAKE_PROJECT_NAME} git_version_generation)

# Source files
file(GLOB APP_SRC ${CMAKE_SOURCE_DIR}/src/app/*.c)
target_sources(${CMAKE_PROJECT_NAME}
    PRIVATE
        ${APP_SRC}
)

# Theses includes are public. You want Zig module to be able to retrieve theses includes if you use them.
target_include_directories(${CMAKE_PROJECT_NAME}
    PUBLIC
        ${CMAKE_BINARY_DIR}
        ${CMAKE_SOURCE_DIR}/Core/Inc
        ${CMAKE_SOURCE_DIR}/src/zig
        ${CMAKE_SOURCE_DIR}/src/app
)

# Link options
set_target_properties(${CMAKE_PROJECT_NAME} PROPERTIES LINK_DEPENDS
    ${CMAKE_SOURCE_DIR}/STM32L476XX_FLASH.ld
)
target_link_options(${CMAKE_PROJECT_NAME} 
    PRIVATE
        -T ${CMAKE_SOURCE_DIR}/STM32L476XX_FLASH.ld
        -Wl,-Map=${CMAKE_PROJECT_NAME}.map,--cref
        -Wl,--print-memory-usage
        -flto
)

# Compile options
target_compile_options(${CMAKE_PROJECT_NAME} PRIVATE
    -Wall
    -Wextra
    -Wshadow
    -Wundef
    -Wcast-qual
    -Wmissing-prototypes
    -Wmissing-include-dirs
    -Wstrict-prototypes
    -Wswitch-default
    -Wswitch-enum
    -Wimplicit-fallthrough
    -Wtype-limits
    -Wno-ignored-qualifiers
    -Werror=strict-prototypes
    -Wpointer-arith
    -Wunreachable-code
    -Wreturn-type
    -Wformat=2
    -Wformat-truncation
    -Wmultichar
    -Wformat-security
    -Wdeprecated
    -Wno-error=cpp
    -Wno-missing-field-initializers
    -Wno-format-nonliteral
    -Wno-cast-qual
    -Wno-switch-default
    -Wno-switch-enum
    -Wno-sign-compare
    -Wdouble-promotion
    -Wconversion
    -Wattributes
    -Wno-comment
)

##############################################################################################
# Zig Library
##############################################################################################
include(${CMAKE_SOURCE_DIR}/cmake/zig-module.cmake)

if(CMAKE_BUILD_TYPE STREQUAL "Debug")
set(ZIG_RELEASE -ODebug -fno-strip)
elseif (CMAKE_BUILD_TYPE STREQUAL "Release")
set(ZIG_RELEASE -OReleaseSafe)
else()
message(FATAL_ERROR "${CMAKE_BUILD_TYPE} is not a valid build type")
endif()

zig_add_module(
    NAME libzig
    LINK_TO ${CMAKE_PROJECT_NAME}
    SRC_ROOT ${CMAKE_SOURCE_DIR}/src/zig/libzig.zig
    TARGET
        -target thumb-freestanding-eabihf
        -mcpu=cortex_m4+vfp4d16sp+dsp
        -fsingle-threaded

    # Optionnal C translation and dependencies
    TRANSLATE_IN ${CMAKE_SOURCE_DIR}/src/zig/translate.h
    TRANSLATE_OUT ${CMAKE_SOURCE_DIR}/src/zig/translate.zig
    TRANSLATE_DEPENDS
        ${CMAKE_BINARY_DIR}/cmake_config.h
        ${CMAKE_BINARY_DIR}/cmake_version.h

    BUILD_OPTS
        ${ZIG_RELEASE}
        
    TRANSLATE_OPTS
        # optional zig flags
)

##############################################################################################
# Generate hex and bin output files
##############################################################################################
add_custom_command(TARGET ${CMAKE_PROJECT_NAME} POST_BUILD
    COMMAND ${CMAKE_OBJCOPY} -O ihex
            $<TARGET_FILE:${CMAKE_PROJECT_NAME}> ${CMAKE_PROJECT_NAME}.hex
    COMMAND ${CMAKE_OBJCOPY} -O binary
            $<TARGET_FILE:${CMAKE_PROJECT_NAME}> ${CMAKE_PROJECT_NAME}.bin
)
